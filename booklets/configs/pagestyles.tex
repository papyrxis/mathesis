% ─────────────────────────────────────────────────────────────────────────────
% Page styles for frontmatter / mainmatter / backmatter
%
% LaTeX's book class calls \thispagestyle{plain} at the start of every
% \chapter and \chapter*, unconditionally — that includes chapter openers,
% and every one of the *matter chapter-style pages (title, how-to-read,
% about-*, references, ...). Left alone, "plain" falls back to a bare
% centered page number in the default font, which is why the branded
% header/footer used to vanish on almost every chapter-opening page (most
% chapters here are short, so that's most of the book). The fix: redefine
% what "plain" means every time \frontmatter / \mainmatter / \backmatter
% is invoked, so it always matches whichever phase is currently active
% instead of falling back to a generic default.
% ─────────────────────────────────────────────────────────────────────────────
\usepackage{fancyhdr}
\usepackage{lastpage}
\usepackage{ifthen}
\usepackage{amssymb}
\usepackage{pdfpages}
\usepackage{wrapfig}

\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

\newcommand{\bkltDiamond}{%
  \kern 0.15em\textcolor{accent}{$\blacklozenge$}\kern 0.15em%
}
\newcommand{\bkltDot}{%
  \kern 0.12em\textcolor{accent!70}{\textperiodcentered}\kern 0.12em%
}

\newcommand{\headerfont}{\sffamily\small\bfseries}
\newcommand{\footerfont}{\sffamily\scriptsize}
\newcommand{\pagenumberfont}{\sffamily\small\bfseries}

\setlength{\headsep}{16pt}

% This is a oneside document — page parity still alternates internally, but
% there are no mirrored margins to justify alternating header content. Every
% page of a given phase gets the same, fixed left/right layout.

% \leftmark holds the full chapter title, which can run arbitrarily long —
% pairing it with the booklet title on the other side risks exactly the
% collision a fixed-width header can't recover from. Keep both sides short
% and length-independent instead: booklet title on the left, chapter number
% on the right. The full chapter title is already on the page itself.
\providecommand{\BookletChapterWord}{Chapter}
\renewcommand{\chaptermark}[1]{\markboth{\thechapter}{}}
\newcommand{\MathesisMainStyle}{%
  \fancyhf{}%
  \fancyhead[L]{\headerfont\color{primary}\textsc{\BookletTitleShort}}%
  \fancyhead[R]{\headerfont\color{accent}\BookletChapterWord~\leftmark}%
  \fancyfoot[C]{%
    \footerfont\color{accent}%
    \bkltDiamond\quad\pagenumberfont\thepage\quad\bkltDiamond%
  }%
  \renewcommand{\headrulewidth}{0.6pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \renewcommand{\headrule}{\hbox to\headwidth{\color{accent}\leaders\hrule height \headrulewidth\hfill}}%
}

\newcommand{\MathesisFrontmatterStyle}{%
  \fancyhf{}%
  \fancyhead[C]{\headerfont\color{primary}\textsc{\BookletTitleShort}}%
  \fancyfoot[C]{\pagenumberfont\color{textsecondary}\thepage}%
  \renewcommand{\headrulewidth}{0.5pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \renewcommand{\headrule}{\hbox to\headwidth{\color{primary!50}\leaders\hrule height 0.45pt\hfill}}%
}

\newcommand{\MathesisBackmatterStyle}{%
  \fancyhf{}%
  \fancyhead[L]{\headerfont\color{primary}\textsc{\BookletTitleShort}}%
  \fancyhead[R]{\headerfont\color{accent}\nouppercase{\leftmark}}%
  \fancyfoot[R]{\pagenumberfont\color{primary}\textbf{\thepage}}%
  \renewcommand{\headrulewidth}{0.8pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

\fancypagestyle{main}{\MathesisMainStyle}
\fancypagestyle{frontmatter}{\MathesisFrontmatterStyle}
\fancypagestyle{backmatter}{\MathesisBackmatterStyle}

% ── EMPTY — cover / intentionally blank ─────────────────────
\fancypagestyle{empty}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
}

\pagestyle{main}

\let\oldfrontmatter\frontmatter
\renewcommand{\frontmatter}{%
  \oldfrontmatter%        ← does \pagenumbering{roman} + resets counter to 1
  \pagestyle{frontmatter}%
  \fancypagestyle{plain}{\MathesisFrontmatterStyle}% chapter-opening pages match too
}

\let\oldmainmatter\mainmatter
\renewcommand{\mainmatter}{%
  \oldmainmatter%         ← does \pagenumbering{arabic} + resets counter to 1
  \pagestyle{main}%
  \fancypagestyle{plain}{\MathesisMainStyle}%
}

\let\oldbackmatter\backmatter
\renewcommand{\backmatter}{%
  \oldbackmatter%         ← no counter reset, continues arabic
  \pagestyle{backmatter}%
  \fancypagestyle{plain}{\MathesisBackmatterStyle}%
}
